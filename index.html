// Supabase Edge Function: proposal-signed-email (sends through Resend — https://resend.com) // Called by TaxBase when a HOD / Executive Administrator enters the "Proposal Signed" date on a CP204 or CP204A row. // The email goes to Tax (ascenddynamic@gmail.com) as proof that the date was recorded, and by whom. // Secrets (Supabase > Edge Functions > Secrets): RESEND_API_KEY (optional: RESEND_FROM, TAX_EMAIL, EXTRA_CC) import { createClient } from "https://esm.sh/@supabase/supabase-js@2.45.4"; const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type" }; const esc = (t: string) => String(t ?? "").replace(/[&<>"]/g, (c) => ({ "&": "&", "<": "<", ">": ">", '"': """ }[c]!)); Deno.serve(async (req) => { if (req.method === "OPTIONS") return new Response("ok", { headers: cors }); try { const url = Deno.env.get("SUPABASE_URL")!; const asUser = createClient(url, Deno.env.get("SUPABASE_ANON_KEY")!, { global: { headers: { Authorization: req.headers.get("Authorization") ?? "" } } }); const { data: { user } } = await asUser.auth.getUser(); if (!user) return new Response("Not signed in", { status: 401, headers: cors }); const admin = createClient(url, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!); const { data: me } = await admin.from("profiles").select("role").eq("id", user.id).single(); if (!me || !["admin", "hod"].includes(me.role)) return new Response("Not allowed", { status: 403, headers: cors }); const b = await req.json(); const to = Deno.env.get("TAX_EMAIL") ?? "ascenddynamic@gmail.com"; const cc = (Deno.env.get("EXTRA_CC") ?? "").split(",").map((x) => x.trim()).filter(Boolean); const from = Deno.env.get("RESEND_FROM") ?? "TaxBase "; const subject = `[TaxBase] ${b.client} signed the proposal — ${b.form}`; const text = `Hi Tax team,\n\n${b.client} has signed the proposal for ${b.form} on ${b.signed_date}.\n\n` + `This change has been made by ${b.changed_by} (${b.changed_role}) on ${b.changed_at}.\n\n` + `This email is a record that the "Proposal Signed" date was entered in TaxBase.\n${b.app_url}\n\n— TaxBase, Ascend Advisory Group`; const html = `

Hi Tax team,

${esc(b.client)} has signed the proposal for ${esc(b.form)} on ${esc(b.signed_date)}.

This change has been made by ${esc(b.changed_by)} (${esc(b.changed_role)}) on ${esc(b.changed_at)}.

This email is a record that the "Proposal Signed" date was entered in TaxBase.

Open TaxBase

— TaxBase, Ascend Advisory Group

`; const r = await fetch("https://api.resend.com/emails", { method: "POST", headers: { Authorization: `Bearer ${Deno.env.get("RESEND_API_KEY")}`, "Content-Type": "application/json" }, body: JSON.stringify({ from, to: [to], cc: cc.length ? cc : undefined, subject, html, text }), }); if (!r.ok) return new Response(await r.text(), { status: 502, headers: cors }); return new Response(JSON.stringify({ ok: true }), { headers: { ...cors, "Content-Type": "application/json" } }); } catch (e) { return new Response(String(e), { status: 500, headers: cors }); } });
